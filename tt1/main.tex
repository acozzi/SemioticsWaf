\section{Introducción}

\subsection{Contexto del proyecto}
\textbf{Contextualizar el escenario general, dominio temático y entorno donde se inserta el proyecto de investigación o desarrollo.}

\vspace{0.2cm}
\noindent \textit{El sistema de bicicletas públicas EcoBici de la Ciudad de Buenos Aires constituye un pilar clave en la movilidad sustentable, registrando miles de desplazamientos diarios entre sus múltiples estaciones distribuidas en la red urbana.}

\subsection{Justificación del estudio}
\textbf{Se debe explicitar la relevancia y conveniencia del estudio planteado, justificando el valor teórico, práctico o metodológico del trabajo.}

\vspace{0.2cm}
\noindent \textit{Comprender la dinámica de flujos y la centralidad de las estaciones permite identificar cuellos de botella, optimizar las estrategias logísticas de rebalanceo de bicicletas y aportar evidencia empírica para la futura expansión de la infraestructura vial urbana \cite{gonzalez2022ecobici}.}

\subsection{Alcances del trabajo}
\textbf{Detallar los límites operacionales y temporales del proyecto (qué abarca concretamente la investigación).}

\vspace{0.2cm}
\noindent \textit{El estudio abarca el análisis del conjunto de datos de viajes de EcoBici correspondientes al período lectivo 2025 en CABA, enfocándose en la topología del grafo, la detección de comunidades urbanas y el cálculo de métricas de centralidad.}

\section{Estado de la cuestión}
\textbf{Se deben estructurar las subsecciones según temáticas relevantes. Aclarar antecedentes de dominio (investigaciones sobre el problema específico) y antecedentes metodológicos (herramientas, algoritmos o técnicas previas aplicadas al área). Utilizar citas bibliográficas mediante BibTeX.}

\subsection{Micromovilidad y sistemas de transporte compartidos}
\noindent \textit{La literatura sobre transporte urbano resalta el uso de sistemas de micro-movilidad para caracterizar la eficiencia del transporte público \cite{gonzalez2022ecobici}. Trabajos previos en redes de bicicletas analizan patrones de demanda temporal y hábitos de desplazamiento en grandes urbes.}

\subsection{Teoría de Redes Complejas}
\noindent \textit{En el ámbito metodológico, la teoría de redes complejas ha sido ampliamente utilizada para modelar grafos dirigidos y ponderados de transporte \cite{newman2018networks}, evaluando propiedades topológicas e identificando nodos críticos de conectividad multimodal \cite{rodriguez2021complex}.}

\section{Definición del problema}

\subsection{Problema general}
\textbf{Describir en detalle la problemática concreta a abordar, sus causas y efectos.}

\vspace{0.2cm}
\noindent \textit{Existe una acusada asimetría en los viajes diarios que genera desequilibrios en la disponibilidad de bicicletas en hora pico, exigiendo costosas operaciones de redistribución sin contar con un modelo predictivo guiado por métricas topológicas de red.}

\subsection{Preguntas de investigación}
\textbf{Formular las preguntas específicas que orientarán la investigación.}

\vspace{0.2cm}
\noindent \textit{¿Cuáles son las estaciones que actúan como conectores críticos (alta intermediación) en la red EcoBici y cómo se agrupan espacialmente las estaciones formando comunidades de alto tráfico?}

\section{Descripción del set de datos}
\label{sec:dataset}

\subsection{Origen y contexto del dataset}
\textbf{Contextualizar explícitamente de dónde proviene el dataset y cómo fue colectado.}

\vspace{0.2cm}
\noindent \textit{Los datos provienen del portal de datos abiertos del Gobierno de la Ciudad de Buenos Aires (BA Data), conteniendo registros anonimizados recolectados automáticamente por los sensores de cada estación de EcoBici.}

\subsection{Descripción técnica}
\textbf{Explicitar las características de las variables e información relevante para el planteamiento de la hipótesis. Esta sección puede incluir gráficas, diagramas u otros recursos descriptivos según corresponda.}

\vspace{0.2cm}
\noindent \textit{El dataset incluye variables estructuradas como ID de estación de origen/destino, fecha/hora inicio y fin, duración y coordenadas. A partir de estas variables se definen operacionalmente los nodos y aristas ponderadas de la matriz de adyacencia.}

\section{Hipótesis}
\textbf{Formular la hipótesis de trabajo. Las variables operativas identificadas para la hipótesis deben haberse detallado previamente en la sección \ref{sec:dataset}.}

\vspace{0.2cm}
\noindent \textit{La red de viajes de EcoBici exhibe propiedades de tipo "mundo pequeño" y una estructura comunitaria altamente modular, donde un subconjunto inferior al 15\% de las estaciones concentra más del 40\% del flujo total de intermediación del sistema.}

\section{Objetivos}

\subsection{Objetivo General}
\textbf{El objetivo general debe ser único en la mayoría de los casos.}

\vspace{0.2cm}
\noindent \textit{Modelar y evaluar la topología de la red de trayectos de EcoBici para identificar nodos críticos y patrones de conectividad en la movilidad urbana de Buenos Aires.}

\subsection{Objetivos Específicos}
\textbf{Los objetivos específicos deben presentarse como títulos en negrita acompañados de descripciones cortas en un formato de lista de ítems numerada.}

\begin{enumerate}
    \item \textit{\textbf{Construcción del grafo de movilidad:} Limpiar y transformar los registros de viajes en un grafo dirigido e interconectado de estaciones.}
    \item \textit{\textbf{Cálculo de métricas de centralidad:} Evaluar indicadores de grado, intermediación (\textit{betweenness}) y cercanía para priorizar nodos estratégicos.}
    \item \textit{\textbf{Detección de comunidades:} Aplicar algoritmos de agrupamiento de grafos para identificar zonas urbanas con alta cohesión de viajes.}
\end{enumerate}

\subsection{Metodología}
\textbf{Detallar el enfoque metodológico, fases de trabajo y procedimientos de análisis. Las actividades detalladas deben estar en correlato con los objetivos específicos y el cronograma de trabajo propuesto.}

\vspace{0.2cm}
\noindent \textit{Se implementará un enfoque cuantitativo basado en la teoría de redes complejas. Se procesarán las matrices de origen-destino y se aplicarán algoritmos de análisis topológico.}

\subsection{Técnicas / Herramientas}
\textbf{Especificar lenguajes, librerías, marcos de trabajo o algoritmos a emplear.}
